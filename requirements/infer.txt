# Build from scratch into a fresh Python 3.11 env (see README, "Rebuild the
# inference env"). Only the top-level packages below are listed: torch,
# transformers, and tokenizers are pulled in and pinned by ms-swift + vllm,
# so we don't pin torch ourselves — doing so conflicts with vllm's own
# torch requirement.
#
# Once the env works, lock exact versions:
#   pip freeze > requirements/infer.lock.txt

vllm                 # models.yaml sets backend: vllm  (infer.py -> swift infer --infer_backend vllm)

flash-linear-attention  # REQUIRED to train Qwen3.5 (train/mhs_sft.py). ms-swift patches
                     # Qwen3_5GatedDeltaNet.forward unconditionally, and its fallback to the
                     # original forward only triggers when 'cu_seq_lens_q' is absent from
                     # kwargs — transformers 5.x passes it, so the patched path always runs
                     # and hard-requires fla's causal_conv1d + chunk_gated_delta_rule.
                     # Without it: ImportError at step 0, even with padding_free=False and
                     # sequence_parallel_size=1. Pure-Triton wheel, no nvcc needed.
                     # NOTE transformers' own is_flash_linear_attention_available() also gates
                     # on is_torch_cuda_available(), so it reads False on a GPU-less login node
                     # even when correctly installed — verify inside a GPU allocation.
